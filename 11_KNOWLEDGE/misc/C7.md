        with pyro.plate("rows", K):

        with pyro.plate("states", K):

## **1) Concrete forward–backward posterior code (Python)**
### **1.1 Gaussian HMM (diagonal covariance) — log-space forward/backward**
```
    import numpy as np
    from dataclasses import dataclass
    from typing import Tuple
    
    EPS = 1e-12
    
    def logsumexp(a: np.ndarray, axis=None, keepdims=False) -> np.ndarray:
        a_max = np.max(a, axis=axis, keepdims=True)
        out = a_max + np.log(np.sum(np.exp(a - a_max), axis=axis, keepdims=True) + EPS)
        if not keepdims:
            out = np.squeeze(out, axis=axis)
        return out
    
    def log_normal_diag(y: np.ndarray, mu: np.ndarray, sigma: np.ndarray) -> np.ndarray:
        """
        y: [T, D]
        mu: [K, D]
        sigma: [K, D]  (std dev > 0)
        returns loglik: [T, K]
        """
        T, D = y.shape
        K = mu.shape[0]
        y_exp = y[:, None, :]          # [T, 1, D]
        mu_exp = mu[None, :, :]        # [1, K, D]
        sig_exp = sigma[None, :, :]    # [1, K, D]
    
        z = (y_exp - mu_exp) / (sig_exp + EPS)
        log_det = np.sum(np.log(sig_exp + EPS), axis=-1)   # [1, K]
        quad = 0.5 * np.sum(z * z, axis=-1)                # [T, K]
        const = 0.5 * D * np.log(2.0 * np.pi)
        return -(const + log_det + quad)                   # [T, K]
    
    @dataclass
    class HMMParams:
        init: np.ndarray   # [K]
        trans: np.ndarray  # [K, K]  trans[i,j] = P(z_t=j | z_{t-1}=i)
        mu: np.ndarray     # [K, D]
        sigma: np.ndarray  # [K, D]
    
    @dataclass
    class HMMPosteriors:
        gamma: np.ndarray   # [T, K]  p(z_t=k | y)
        xi: np.ndarray      # [T-1, K, K]  p(z_t=i, z_{t+1}=j | y)
        loglik: float
    
    def forward_backward(y: np.ndarray, params: HMMParams) -> HMMPosteriors:
        """
        Computes smoothed posteriors via log-space forward-backward.
        """
        init = np.clip(params.init, EPS, 1.0)
        trans = np.clip(params.trans, EPS, 1.0)
        init = init / init.sum()
        trans = trans / trans.sum(axis=1, keepdims=True)
    
        T, D = y.shape
        K = init.shape[0]
    
        log_init = np.log(init)
        log_trans = np.log(trans)
        log_emiss = log_normal_diag(y, params.mu, params.sigma)  # [T, K]
    
        # Forward: log_alpha[t,k] = log p(y_0..y_t, z_t=k)
        log_alpha = np.zeros((T, K))
        log_alpha[0] = log_init + log_emiss[0]
        for t in range(1, T):
            # log_alpha[t,k] = log_emiss[t,k] + logsum_j log_alpha[t-1,j] + log_trans[j,k]
            tmp = log_alpha[t-1][:, None] + log_trans  # [K, K]
            log_alpha[t] = log_emiss[t] + logsumexp(tmp, axis=0)
    
        loglik = float(logsumexp(log_alpha[T-1], axis=0))
    
        # Backward: log_beta[t,k] = log p(y_{t+1}..y_{T-1} | z_t=k)
        log_beta = np.zeros((T, K))
        log_beta[T-1] = 0.0
        for t in range(T-2, -1, -1):
            # log_beta[t,i] = logsum_j log_trans[i,j] + log_emiss[t+1,j] + log_beta[t+1,j]
            tmp = log_trans + (log_emiss[t+1] + log_beta[t+1])[None, :]  # [K, K]
            log_beta[t] = logsumexp(tmp, axis=1)
    
        # gamma[t,k] ∝ exp(log_alpha[t,k] + log_beta[t,k])
        log_gamma = log_alpha + log_beta
        log_gamma = log_gamma - logsumexp(log_gamma, axis=1, keepdims=True)
        gamma = np.exp(log_gamma)  # [T, K]
    
        # xi[t,i,j] ∝ alpha[t,i] * trans[i,j] * emiss[t+1,j] * beta[t+1,j]
        xi = np.zeros((T-1, K, K))
        for t in range(T-1):
            log_xi_t = (
                log_alpha[t][:, None]
                + log_trans
                + log_emiss[t+1][None, :]
                + log_beta[t+1][None, :]
            )  # [K, K]
            log_xi_t = log_xi_t - logsumexp(log_xi_t, axis=(0, 1))
            xi[t] = np.exp(log_xi_t)
    
        return HMMPosteriors(gamma=gamma, xi=xi, loglik=loglik)

### **1.2 Dwell times + hazard-of-exit (state persistence / “stuck” detection)**
**Two ways** to estimate “hazard of exit”:
### **(A) Markov hazard from transition matrix (model-based, simple)**
For state k, per-step exit probability:
  * h_k = 1 - P(z_{t+1}=k \mid z_t=k) = 1 - A_{kk}


Expected dwell length (geometric):
  * E[L_k] = 1 / (1 - A_{kk})


### **(B) Empirical hazard conditioned on dwell duration (data-driven)**
Compute survival curve of runs in each state from a decoded or posterior sample.

          hazard[k] = 1 - trans[k,k]
          expected_dwell[k] = 1 / hazard[k]
        """
        K = trans.shape[0]
        hazard = 1.0 - np.clip(np.diag(trans), 0.0, 1.0)
        hazard = np.maximum(hazard, EPS)
        expected_dwell = 1.0 / hazard
        return hazard, expected_dwell
    
    def viterbi_decode(y: np.ndarray, params: HMMParams) -> np.ndarray:
        """
        Simple Viterbi decode for most likely state sequence.
        """
        init = np.clip(params.init, EPS, 1.0)
        trans = np.clip(params.trans, EPS, 1.0)
        init = init / init.sum()
        trans = trans / trans.sum(axis=1, keepdims=True)
    
        T, D = y.shape
        K = init.shape[0]
        log_init = np.log(init)
        log_trans = np.log(trans)
        log_emiss = log_normal_diag(y, params.mu, params.sigma)  # [T,K]
    
        dp = np.zeros((T, K))
        ptr = np.zeros((T, K), dtype=int)
    
        dp[0] = log_init + log_emiss[0]
        for t in range(1, T):
            scores = dp[t-1][:, None] + log_trans  # [K,K]
            ptr[t] = np.argmax(scores, axis=0)
            dp[t] = log_emiss[t] + np.max(scores, axis=0)
    
        z = np.zeros(T, dtype=int)
        z[T-1] = int(np.argmax(dp[T-1]))
        for t in range(T-2, -1, -1):
            z[t] = ptr[t+1, z[t+1]]
        return z
    
    def run_lengths(z: np.ndarray) -> Dict[int, List[int]]:
        """
        Returns list of run lengths for each state.
        """
        runs: Dict[int, List[int]] = {}
        if len(z) == 0:
            return runs
        curr = z[0]
        length = 1
        for t in range(1, len(z)):
            if z[t] == curr:
                length += 1
            else:
                runs.setdefault(int(curr), []).append(length)
                curr = z[t]
                length = 1
        runs.setdefault(int(curr), []).append(length)
        return runs
    
    def empirical_hazard_by_duration(runs: Dict[int, List[int]], max_d: int = 30) -> Dict[int, np.ndarray]:
        """
        Empirical discrete-time hazard h(d) = P(exit at d | survived to d)
        where d = 1..max_d (run length in steps).
        """
        hazards = {}
        for k, lengths in runs.items():
            lengths = np.array(lengths, dtype=int)
            # survival count at duration d: number of runs with length >= d
            surv = np.array([(lengths >= d).sum() for d in range(1, max_d+1)], dtype=float)
            # exit count at duration d: number of runs with length == d
            exitc = np.array([(lengths == d).sum() for d in range(1, max_d+1)], dtype=float)
            h = np.where(surv > 0, exitc / surv, 0.0)
            hazards[k] = h
        return hazards
    
    def posterior_expected_dwell(gamma: np.ndarray) -> np.ndarray:
        """
        Rough posterior expected dwell proxy:
        expected consecutive occupancy can be approximated by state self-transition
        estimated from xi / gamma if you have xi.
        This function just returns mean occupancy; use xi-based estimate for transitions.
        """
        return gamma.mean(axis=0)
```

### **3.3 Fit HMM (simple EM) + infer posteriors**
You can either:

Below is a **minimal EM loop** for diagonal Gaussian HMM:
```
    # src/models/hmm.py
    import numpy as np
    from typing import Tuple
    from .fb import forward_backward, HMMParams  # assume you placed code from section 1 in fb.py
    
    def em_fit_hmm(y: np.ndarray, K: int, iters: int = 50, seed: int = 0) -> HMMParams:
        """
        EM for Gaussian HMM with diagonal covariance.
        """
        rng = np.random.default_rng(seed)
        T, D = y.shape
    
        # init with random responsibilities
        resp = rng.dirichlet(np.ones(K), size=T)  # [T,K]
    
        init = resp[0] / resp[0].sum()
        trans = np.ones((K, K)) / K
    
        mu = (resp.T @ y) / (resp.sum(axis=0)[:, None] + 1e-9)  # [K,D]
        var = np.zeros((K, D))
        for k in range(K):
            diff = y - mu[k]
            var[k] = (resp[:, k][:, None] * diff * diff).sum(axis=0) / (resp[:, k].sum() + 1e-9)
        sigma = np.sqrt(var + 1e-6)
    
        params = HMMParams(init=init, trans=trans, mu=mu, sigma=sigma)
    
        for _ in range(iters):
            post = forward_backward(y, params)
            gamma, xi = post.gamma, post.xi
    
            # M-step
            init = gamma[0]
            trans = xi.sum(axis=0)  # [K,K]
            trans = trans / (trans.sum(axis=1, keepdims=True) + 1e-12)
    
            mu = (gamma.T @ y) / (gamma.sum(axis=0)[:, None] + 1e-12)
            var = np.zeros((K, D))
            for k in range(K):
                diff = y - mu[k]
                var[k] = (gamma[:, k][:, None] * diff * diff).sum(axis=0) / (gamma[:, k].sum() + 1e-12)
            sigma = np.sqrt(var + 1e-6)
    
            params = HMMParams(init=init, trans=trans, mu=mu, sigma=sigma)
    
        return params

def log_normal_diag(y: np.ndarray, mu: np.ndarray, sigma: np.ndarray) -> np.ndarray:  
"""  
y: [T, D]  
mu: [K, D]  
sigma: [K, D] (std dev)  
returns loglik: [T, K]  
"""  
T, D = y.shape  
K = mu.shape[0]
```

    mu_exp = mu[None, :, :]      # [1,K,D]
    sig_exp = sigma[None, :, :]  # [1,K,D]
    
    z = (y_exp - mu_exp) / (sig_exp + EPS)
    log_det = np.sum(np.log(sig_exp + EPS), axis=-1)  # [1,K]
    quad = 0.5 * np.sum(z * z, axis=-1)               # [T,K]
    const = 0.5 * D * np.log(2.0 * np.pi)
    return -(const + log_det + quad)

    log_beta[T - 1] = 0.0
    for t in range(T - 2, -1, -1):
        tmp = log_trans + (log_emiss[t + 1] + log_beta[t + 1])[None, :]  # [K,K]
        log_beta[t] = logsumexp(tmp, axis=1)

    log_gamma -= logsumexp(log_gamma, axis=1, keepdims=True)

            + log_trans
            + log_emiss[t + 1][None, :]
            + log_beta[t + 1][None, :]
        )
        log_xi_t -= logsumexp(log_xi_t, axis=(0, 1))
        xi[t] = np.exp(log_xi_t)
