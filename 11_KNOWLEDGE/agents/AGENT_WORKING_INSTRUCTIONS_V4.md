### **🛡️ Adding Governance**

# 1. Initialize governance
from memory_governance import MemoryGovernanceSystem
from policy_gate import PolicyGateSystem

memory_gov = MemoryGovernanceSystem()
policy_gate = PolicyGateSystem()

# 2. Process with governance
async def process_with_governance(operation, data):
    # Check policy gate
    policy_request = PolicyRequest(
        request_id=f"req_{int(time.time())}",
        operation=operation,
        resource="system_resource",
        user_id="amos_system",
        context={"input_type": type(data).__name__}
    )
    
    evaluations = await policy_gate.evaluate_request(policy_request)
    final_action = policy_gate.get_final_action(evaluations)
    
    if final_action.value == "deny":
        return {
            "success": False,
            "error": "Policy denied",
            "policy_action": final_action.value
        }
    
    # Process operation
    result = await process_operation(operation, data)
    
    # Store in memory governance
    memory_gov.store_memory(
        str(result),
        memory_gov.MemoryClass.WORKING_SESSION,
        ["operation", "governed"],
        {"operation": operation, "policy_action": final_action.value}
    )
    
    return result
